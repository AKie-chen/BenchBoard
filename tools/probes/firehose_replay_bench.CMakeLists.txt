cmake_minimum_required(VERSION 3.19)
project(firehose_replay_bench LANGUAGES CXX)
# --- 仓库根目录 ---------------------------------------------------------
# 默认从本文件位置反推（在 tools/probes/ 就地构建时用）。
# xxx_run.sh 会把探针源码复制到临时目录再构建，那时由脚本传 -DBENCHBOARD_ROOT= 覆盖。
if(NOT DEFINED BENCHBOARD_ROOT)
  get_filename_component(BENCHBOARD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_compile_definitions("BENCHBOARD_ROOT=\"${BENCHBOARD_ROOT}\"")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

if (MSVC)
  add_compile_options(/utf-8 /W4)
endif()

# 只用 QByteArray / QFile / QElapsedTimer -> 只要 Qt6::Core
find_package(Qt6 REQUIRED COMPONENTS Core)

set(USER_SRC "${BENCHBOARD_ROOT}/src")
add_executable(firehose_replay_bench main.cpp
  ${USER_SRC}/MetricsAggregator.cpp ${USER_SRC}/MetricsAggregator.h)
target_include_directories(firehose_replay_bench PRIVATE ${USER_SRC})
target_link_libraries(firehose_replay_bench PRIVATE Qt6::Core)
