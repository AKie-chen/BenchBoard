cmake_minimum_required(VERSION 3.19)
project(firehose_ui LANGUAGES CXX)
# --- 仓库根目录 ---------------------------------------------------------
# 默认从本文件位置反推（在 tools/probes/ 就地构建时用）。
# xxx_run.sh 会把探针源码复制到临时目录再构建，那时由脚本传 -DBENCHBOARD_ROOT= 覆盖。
if(NOT DEFINED BENCHBOARD_ROOT)
  get_filename_component(BENCHBOARD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_compile_definitions("BENCHBOARD_ROOT=\"${BENCHBOARD_ROOT}\"")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

if (MSVC)
  add_compile_options(/utf-8 /W4)
endif()

# ★ 构建类型由 run.sh 传进来，必须能切 Debug。
#   默认的 Debug 是为了和 apps 侧（tools/build.sh）对齐 ——
#   两者性能差 20 倍以上，探针用错构建类型会得出相反结论。
find_package(Qt6 REQUIRED COMPONENTS Widgets Charts)

set(USER_SRC "${BENCHBOARD_ROOT}/src")
add_executable(firehose_ui main.cpp
  ${USER_SRC}/MainWindow.cpp ${USER_SRC}/MainWindow.h
  ${USER_SRC}/MetricsAggregator.cpp ${USER_SRC}/MetricsAggregator.h
  ${USER_SRC}/RealtimeChartView.cpp ${USER_SRC}/RealtimeChartView.h
  ${USER_SRC}/SummaryParser.cpp ${USER_SRC}/SummaryParser.h
  ${USER_SRC}/types.h)
target_include_directories(firehose_ui PRIVATE ${USER_SRC})
target_link_libraries(firehose_ui PRIVATE Qt6::Widgets Qt6::Charts)
