cmake_minimum_required(VERSION 3.19)
project(m4window LANGUAGES CXX)
# --- 仓库根目录 ---------------------------------------------------------
# 默认从本文件位置反推（在 tools/probes/ 就地构建时用）。
# xxx_run.sh 会把探针源码复制到临时目录再构建，那时由脚本传 -DBENCHBOARD_ROOT= 覆盖。
if(NOT DEFINED BENCHBOARD_ROOT)
  get_filename_component(BENCHBOARD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_compile_definitions("BENCHBOARD_ROOT=\"${BENCHBOARD_ROOT}\"")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

if (MSVC)
  add_compile_options(/utf-8 /W4)
endif()

find_package(Qt6 REQUIRED COMPONENTS Widgets Charts)

set(USER_SRC "${BENCHBOARD_ROOT}/src")

# ★ 这里链接【真实的】src/ 实现 —— 探针的目的是验收它们，不是模拟它们。
#   （与 m4_skeleton_check 相反：那个是"下发前自测 TODO 能不能落地"，
#     所以在探针里自己实现了一遍；这个是"验收用户写的东西"，必须用原件。）
add_executable(m4window m4_window_probe.cpp
  ${USER_SRC}/MetricsAggregator.cpp
  ${USER_SRC}/MetricsAggregator.h
  ${USER_SRC}/RealtimeChartView.cpp
  ${USER_SRC}/RealtimeChartView.h
  ${USER_SRC}/types.h)

target_include_directories(m4window PRIVATE ${USER_SRC})
target_link_libraries(m4window PRIVATE Qt6::Widgets Qt6::Charts)
