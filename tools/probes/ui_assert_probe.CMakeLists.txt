cmake_minimum_required(VERSION 3.19)
project(uiprobe LANGUAGES CXX)
# --- 仓库根目录 ---------------------------------------------------------
# 默认从本文件位置反推（在 tools/probes/ 就地构建时用）。
# xxx_run.sh 会把探针源码复制到临时目录再构建，那时由脚本传 -DBENCHBOARD_ROOT= 覆盖。
if(NOT DEFINED BENCHBOARD_ROOT)
  get_filename_component(BENCHBOARD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_compile_definitions("BENCHBOARD_ROOT=\"${BENCHBOARD_ROOT}\"")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if (MSVC)
  add_compile_options(/utf-8)
endif()

find_package(Qt6 REQUIRED COMPONENTS Widgets Charts)

# 直接引用用户源码，不复制副本 —— 验证的就是他真正写的东西
set(USER_SRC "${BENCHBOARD_ROOT}/src")
add_executable(uiprobe main.cpp ${USERSRC}/MainWindow.cpp ${USERSRC}/MainWindow.h
  ${USERSRC}/MetricsAggregator.cpp ${USERSRC}/MetricsAggregator.h
  ${USERSRC}/RealtimeChartView.cpp ${USERSRC}/RealtimeChartView.h
  ${USERSRC}/SummaryParser.cpp ${USERSRC}/SummaryParser.h
  ${USERSRC}/types.h)
target_include_directories(uiprobe PRIVATE ${USERSRC})
target_link_libraries(uiprobe PRIVATE Qt6::Widgets Qt6::Charts)
