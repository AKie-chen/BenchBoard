cmake_minimum_required(VERSION 3.19)
project(qprocess_startdiag LANGUAGES CXX)
# --- 仓库根目录 ---------------------------------------------------------
# 默认从本文件位置反推（在 tools/probes/ 就地构建时用）。
# xxx_run.sh 会把探针源码复制到临时目录再构建，那时由脚本传 -DBENCHBOARD_ROOT= 覆盖。
if(NOT DEFINED BENCHBOARD_ROOT)
  get_filename_component(BENCHBOARD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_compile_definitions("BENCHBOARD_ROOT=\"${BENCHBOARD_ROOT}\"")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC OFF)

find_package(Qt6 REQUIRED COMPONENTS Core)

add_executable(qprocess_startdiag main.cpp)
target_link_libraries(qprocess_startdiag PRIVATE Qt6::Core)

if (MSVC)
  target_compile_options(qprocess_startdiag PRIVATE /utf-8 /W4)
endif()
